cmake_minimum_required(VERSION 3.20)
project(ScreenRelay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "ScreenRelay requires Windows")
endif()
add_executable(ScreenRelay WIN32 src/app.cpp)
target_compile_definitions(ScreenRelay PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(ScreenRelay PRIVATE winhttp ws2_32 crypt32 shell32 user32 advapi32 ole32 oleaut32 taskschd uuid)
if(MSVC)
  set_property(TARGET ScreenRelay PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ScreenRelay PRIVATE /W4 /utf-8)
else()
  target_link_options(ScreenRelay PRIVATE -static -municode)
  target_compile_options(ScreenRelay PRIVATE -Wall -Wextra)
endif()
enable_testing()
foreach(test controller network display)
  add_executable(${test}_test tests/${test}_test.cpp)
  target_compile_definitions(${test}_test PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(${test}_test PRIVATE winhttp ws2_32 crypt32 user32)
  if(MINGW)
    target_link_options(${test}_test PRIVATE -static)
  endif()
  add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
